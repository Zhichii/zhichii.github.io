{{- /* 按 Minecraft 快照的日期写法输出：26w10f · 066 · 2026 年 3 月 7 日 · 26 两位年份（取该周星期四所在的年份，即 ISO 年） · w10 ISO 周数 · f 星期几，a=周一 … g=周日 · 066 年内第几天，元旦记作 001 */ -}} {{- $d := .Date -}} {{- $wd := int $d.Weekday -}} {{- $wdISO := add 1 (mod (add $wd 6) 7) -}} {{- $thu := $d.AddDate 0 0 (sub 4 $wdISO) -}} {{- $week := add 1 (div (sub $thu.YearDay 1) 7) -}} {{- $letter := index (slice "a" "b" "c" "d" "e" "f" "g") (sub $wdISO 1) -}} {{- printf "%sw%02d%s" ($thu.Format "06") $week $letter }} · {{ printf "%03d" $d.YearDay }} · {{ $d.Format "2006 年 1 月 2 日" -}}
